\begin{frame}{Problem: Neural Network Extrapolation}
    \begin{itemize}
        \item Neural Networks often struggle with extrapolation beyond training data
        \item We like ReLU
        \begin{enumerate}
            \item Natural linear extrapolation capability - unbounded
            \item Non-differentiable at x = 0 ("0-points")
            \item Cheap to compute
        \end{enumerate}
        \item Two key issues with ReLU networks:
        \begin{enumerate}
            \item Non-linearities outside training range produce no learning
            \item High variation in model derivative
        \end{enumerate}
    \end{itemize}
\end{frame}

\begin{frame}{ReLEx - Lopedoto et al., 2020}
    \begin{table}[t!]
     \centering
     \begin{tabular}{llcc} \hline
     Loss Name  & Definition & Complexity \\
     \hline \hline
     CP - Centripetal  & $\sum_{h=1}^H \sum_{k=1}^K (x_{h0} - x_{k})^2$ & $O(H K)$ \\
     \hfill
     \\
     MR - Mutually Repellent & $\sum_{h=1}^H \sum_{j>i}^H \frac{1}{(x_{h0} - x_{j0})^2+ \varepsilon}$ & $O(H^2)$ \\ 
     \hfill
     \\
     WO - Weight Orientation & $- \sum_{h=1}^H {ReLU((x_{h0} - \bar{x})\cdot w_h + \varepsilon)}$ & $O(H)$ \\
     \hfill
     \\
     WS - Weight Sign $ & $({\sum_{h=1}^Hw_h})^2$ & $O(H)$ 
     \\
     \hline
    \end{tabular}
    
    \caption{The loss terms used in ReLEx improve extrapolation. 
    $K$ is the number of data points.
    $H$ is the number of hidden neurons. 
    $x_{h0}$ is the \emph{0-point} of hidden neuron $h$ (dependent only on the network weights $w_h$ and $b_h$).
    $\bar{x}$ is the average of the training data. 
    The complexity refers to a single calculation of the loss value. 
    }
     
    \end{table}
\end{frame}

\begin{frame}{The DATE Method}
    Two new regularization terms:
    \begin{enumerate}
        \item IR-loss ($L_{IR}$):
        \begin{itemize}
            \item Keeps 0-points inside training data range
            \item More efficient than ReLEx approach
        \end{itemize}
        \item D1-loss ($L_{D1}$):
        \begin{itemize}
            \item Penalizes deviation of model derivative from target derivative
            \item Estimated from training data points
        \end{itemize}
        
    \end{enumerate}
\end{frame}

\begin{frame}{IR-Loss Implementation}
    \begin{itemize}
        \item Purpose: Keep 0-points within training range
        \item Formula:
        \[ L_{IR} = \sum_{k=1}^K (ReLU(-x_{k0} + L - \tau) + ReLU(x_{k0} - R + \tau))^2 \]
        where:
        \begin{itemize}
            \item $L$, $R$: Left and right bounds of training range
            \item $\tau$: Margin parameter
            \item $x_{k0}$: 0-point of neuron k
        \end{itemize}
        \item Complexity: O(K) vs O(H²) in ReLEx
    \end{itemize}
\end{frame}

\begin{frame}{D1-Loss Implementation}
    \begin{itemize}
        \item Derivative estimation:
        \[ y'_k = \frac{y_{k+1} - y_k}{x_{k+1} - x_k} \]
        \item Model derivative:
        \[ \hat{f}'(x) = \frac{\hat{f}(x + \epsilon) - \hat{f}(x - \epsilon)}{2\epsilon} \]
        \item D1-loss:
        \[ L_{D1} = \sum_{(x',y')\in XY'} (y' - \hat{f}'(x'))^2 \]
    \end{itemize}
\end{frame}

\begin{frame}{Experimental Setup}
    \begin{itemize}
        \item Network: Single hidden layer, 10 neurons
        \item ReLU activation: $ReLU(x) = max(0,x)$

        \item Training data:
        \begin{itemize}
            \item Range: [-2, 2]
            \item 200 equidistant points
            \item Test functions: linear, absolute, quadratic, sigmoid, sine
        \end{itemize}
        \item Hyperparameters:
        \begin{itemize}
            \item Learning rate: 0.007
            \item Momentum: 0.9
            \item 700 epochs, batch size 20
        \end{itemize}
    \end{itemize}
\end{frame}


\begin{frame}{Evaluation Metrics}
    \begin{itemize}
        \item Interpolation metrics:
        \begin{itemize}
            \item $MSE_{int}$: Error in training range
            \item $MSE_{D1}$: Derivative error in training range
        \end{itemize}
        \item Extrapolation metrics:
        \begin{itemize}
            \item $Tan_{ex}$: Error vs tangent lines
            \item $D1_{ex}$: Error vs last two points slope
            \item $WLR_{ex}$: Error vs weighted linear regression
            
        \end{itemize}
        \item Consistency: $\sigma_S$ (slope standard deviation)
    \end{itemize}
\end{frame}


\begin{frame}{DATE Configurations}
    Three variants tested:
    \begin{enumerate}
        \item $DATE_1$: Full ReLEx + $L_{D1}$
        \item $DATE_2$: $L_{WO} + L_{WS} + L_{IR} + L_{D1}$
        \item $DATE_3$: $L_{WS} + L_{IR} + L_{D1}$
    \end{enumerate}
    Key differences:
    \begin{itemize}
        \item $DATE_1$: Most complex but stable
        \item $DATE_2$: Balanced performance
        \item $DATE_3$: Simplest configuration
    \end{itemize}
\end{frame}

\begin{frame}{Conclusion}
    \begin{itemize}
        \item DATE improves extrapolation while maintaining good interpolation
        \item Key advantages:
        \begin{itemize}
            \item Better derivative alignment
            \item More efficient than ReLEx
            \item Multiple configuration options
        \end{itemize}
    
    \end{itemize}
\end{frame}


\begin{frame}{Improvements \& Future Work}
    \begin{itemize}
        \item Current limitations:
        \begin{itemize}
            \item Only univariate regression
            \item Single hidden layer
            \item Assumes linear extrapolation is desired
            \item Sensitivity to noise at margins
        \end{itemize}
        \item Future work:
        \begin{itemize}
            \item Extension to multiple dimensions
            \item Different network architectures
            \item Higher-order derivatives
            \item Working with Smoothing
        \end{itemize}
    \end{itemize}
\end{frame}
